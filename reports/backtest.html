# =============================================================================
#  米国株 バックテスト設定
#
#  ここを書き換えるだけで戦略を検証できる。Pythonのコードは触らなくていい。
#  「#」から後ろはコメント（実行に影響しない）。
#
#  実行:  python backtest.py
# =============================================================================

name: "米国株_押し目買い_v1"

# -----------------------------------------------------------------------------
#  1. データ
# -----------------------------------------------------------------------------
data:
  source: yfinance
  start: "2015-01-01"
  end:   "2025-12-31"

  # 銘柄は universe.json から読む。build_universe.py で作る。
  #   sp500  … S&P500の大型株（約500銘柄）
  #   midcap … 流動性フィルタを通した中型株
  #   etf    … 主要ETFとセクターETF
  #   all    … 全部
  universe: sp500
  universe_limit: null      # 数字を入れると先頭N銘柄だけで試せる（動作確認用）

  benchmark: "SPY"

# -----------------------------------------------------------------------------
#  2. 売買ルール
#
#  使える部品:
#    価格      : open, high, low, close, volume, typical
#    移動平均  : sma(close,50)  ema(close,20)
#    オシレータ: rsi(close,14)  macd_hist(close)  stoch_k(high,low,close,14)
#    ボラ      : atr_pct(high,low,close,14)  hist_vol(close,20)  stdev(close,20)
#    ボリンジャ: bb_lower(close,20,2)  bb_pctb(close,20,2)
#    位置      : highest(close,60)  lowest(close,60)  pct_from_high(close,252)  zscore(close,10)
#    勢い      : roc(close,120)  slope(close,20)  rising(close,3)
#    交差      : cross_above(a,b)  cross_below(a,b)
#    過去参照  : shift(close,1)   ※ 未来参照は文法レベルで禁止
# -----------------------------------------------------------------------------
rules:
  # 長期上昇トレンドの中で、いったん売られたあと反発し始めたところを拾う。
  # 米国株で最も再現性が確認されている「短期反転」を素直に使った形。
  entry: >
    close > sma(close, {ma_trend})
    and cross_above(rsi(close, {rsi_n}), {rsi_buy})
    and atr_pct(high, low, close, 14) < {max_vol}

  # cross_below を使うのが要点。「close < sma」だと建てた瞬間に条件を満たしてしまう。
  exit: >
    rsi(close, {rsi_n}) > {rsi_sell}
    or cross_below(close, sma(close, {ma_exit}))

  # 候補が枠より多いときの優先順位。
  # 直近1ヶ月を除いた12ヶ月モメンタムが強い順（短期反転の影響を除く定番の作り方）。
  rank: "roc(shift(close, 21), 231)"
  rank_ascending: false

  # 流動性フィルタ。30万円程度でも板の薄い銘柄は避ける。
  universe_filter: "sma(volume, 20) * close > 20000000"

  # 相場全体のフィルタ。SPYが200日線を下回る間は新規建てを止める。
  # この1行だけで最大ドローダウンが大きく改善することが多い。
  market_filter: "close > sma(close, 200)"

  stop_loss_pct:   0.08
  take_profit_pct: null
  trail_stop_pct:  null
  max_hold_days:   20

# -----------------------------------------------------------------------------
#  3. パラメータ
# -----------------------------------------------------------------------------
params:
  ma_trend: 200
  ma_exit:  20
  rsi_n:    14
  rsi_buy:  35
  rsi_sell: 70
  max_vol:  0.06        # 1日のATRが株価の6%を超える銘柄は避ける

# -----------------------------------------------------------------------------
#  4. 資金管理
#     30万円 ≒ 2,000ドル。米国株は1株単位なので少額でも分散が効く。
# -----------------------------------------------------------------------------
portfolio:
  initial_cash:  2000
  max_positions: 8
  position_pct:  null
  lot_size:      1        # 米国株は1株単位

  # 端株（フラクショナル株）を許可する。少額運用では実質必須。
  # 2,000ドルを8銘柄に分けると1銘柄250ドルなので、これが無いと
  # 株価250ドル超の銘柄（MSFT, AVGO等）が最初から買えなくなる。
  allow_fractional: true

# -----------------------------------------------------------------------------
#  5. 取引コスト
#     米国株は手数料無料の証券会社が多いが、スプレッドと滑りは必ず発生する。
#     ここをゼロにした検証結果は信じてはいけない。
# -----------------------------------------------------------------------------
costs:
  commission_bps: 0.0
  slippage_bps:   8.0     # 片道0.08%。大型株の寄付成行の実感値
  min_commission: 0

# -----------------------------------------------------------------------------
#  6. 情報源（世界情勢のRSS）
#
#     個別銘柄に紐づかない材料 — 金融政策、地政学、政策変更 — を拾う。
#     ここを書き換えれば、追いたいニュース源を自由に足せる。
#     形式は [URL, 表示名] の並び。APIキーは不要。
# -----------------------------------------------------------------------------
feeds:
  rss:
    - ["https://www.federalreserve.gov/feeds/press_all.xml", "FRB発表"]
    - ["https://home.treasury.gov/system/files/126/press-releases.xml", "米財務省"]
    - ["https://feeds.a.dj.com/rss/RSSWorldNews.xml", "世界情勢"]
    - ["https://feeds.a.dj.com/rss/RSSMarketsMain.xml", "市場"]
    - ["https://feeds.a.dj.com/rss/RSSWSJD.xml", "テクノロジー"]
    - ["https://www.ecb.europa.eu/rss/press.html", "ECB発表"]

  # SEC EDGARの全文検索で追う語。開示の「本文」に出てきたら拾う。
  # 業績見通しの変更、減損、リストラは株価が動く前触れになることが多い。
  edgar_queries:
    - "lowered guidance"
    - "impairment charge"
    - "restructuring plan"
    - "material weakness"
    - "going concern"

  # Redditの言及数を集計するか。非商用なら無料。
  # 言及数そのものではなく「急増」を異常検知として使う。
  reddit: true

# -----------------------------------------------------------------------------
#  7. 検証方法
# -----------------------------------------------------------------------------
validation:
  split_date: "2022-01-01"
  monte_carlo: true
  walk_forward:
    train_years: 4
    test_years:  1

# -----------------------------------------------------------------------------
#  8. パラメータ感度の調査
#
#     max_hold_days をここに入れてあるのが重要。
#     「何日保有すべきか」を思い込みで決めず、同じデータで全部比較する。
# -----------------------------------------------------------------------------
param_grid:
  ma_trend:      [100, 200]
  ma_exit:       [10, 20, 50]
  rsi_n:         [14]
  rsi_buy:       [25, 30, 35, 40]
  rsi_sell:      [65, 70, 75]
  max_vol:       [0.06]
  max_hold_days: [3, 5, 10, 20, 60]
